import {NextRequest,NextResponse} from 'next/server';
import {parseUrl,mangaInfo,chapters,pageUrls} from '@/lib/sources';
export async function POST(req:NextRequest){
  const{url}=await req.json(); const p:any=parseUrl(url);
  if(p.status!=='supported')return NextResponse.json(p,{status:422});
  try{
    let mangaId=p.mangaId;
    if(!mangaId&&p.chapterId){const r=await(await fetch(`https://api.mangadex.org/chapter/${p.chapterId}`)).json();
      mangaId=r.data.relationships.find((x:any)=>x.type==='manga').id}
    const[info,chs]=await Promise.all([mangaInfo(mangaId),chapters(mangaId)]);
    const pages=p.chapterId?await pageUrls(p.chapterId):[];
    return NextResponse.json({status:'supported',source:p.source,info,chapters:chs,selected:p.chapterId,pages});
  }catch(e:any){return NextResponse.json({status:'connection_error',error:e.message+'. Coba lagi atau pakai upload.'},{status:502})}
}

import {NextRequest,NextResponse} from 'next/server';
import OpenAI from 'openai';
import {createHash} from 'crypto';
const oa=new OpenAI(); // OPENAI_API_KEY (server only)
const cache=new Map<string,Buffer>();
const VOICES=['alloy','ash','ballad','coral','echo','fable','nova','onyx','sage','shimmer'];
const STYLE:Record<string,string>={happy:'cheerful',sad:'sorrowful, slow',angry:'angry, sharp',whispering:'whispering softly',surprised:'startled',confident:'confident',scared:'trembling, fearful',serious:'serious, low',excited:'excited, fast',sarcastic:'sarcastic',crying:'crying, voice breaking',calm:'calm, even'};
export async function POST(req:NextRequest){
  const{text,voice='alloy',emotion='calm',speed=1}=await req.json();
  if(!text||!VOICES.includes(voice))return NextResponse.json({error:'Parameter TTS tidak valid.'},{status:400});
  const h=createHash('sha256').update([text,voice,emotion,speed].join('|')).digest('hex');
  let buf=cache.get(h);
  try{
    if(!buf){const r=await oa.audio.speech.create({model:'gpt-4o-mini-tts',voice,input:text,
      speed:Math.min(4,Math.max(0.25,speed)),instructions:`Baca seperti aktor suara anime. Gaya: ${STYLE[emotion]||'natural'}.`} as any);
      buf=Buffer.from(await r.arrayBuffer());if(cache.size>300)cache.delete(cache.keys().next().value!);cache.set(h,buf)}
    return new NextResponse(buf,{headers:{'Content-Type':'audio/mpeg'}});
  }catch(e:any){return NextResponse.json({error:'TTS gagal: '+e.message},{status:502})}
}

// Adapter per sumber. Hanya sumber ber-API resmi; tidak ada bypass proteksi.
export type Status = 'supported'|'login_required'|'unsupported'|'connection_error';
const UUID='[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}';
const API='https://api.mangadex.org';
export function parseUrl(raw:string){
  let u:URL; try{u=new URL(raw)}catch{return{status:'unsupported' as Status,error:'URL tidak valid.'}}
  if(/(^|\.)mangadex\.org$/.test(u.hostname)){
    const t=u.pathname.match(new RegExp(`/title/(${UUID})`)),c=u.pathname.match(new RegExp(`/chapter/(${UUID})`));
    if(t||c)return{status:'supported' as Status,source:'mangadex',mangaId:t?.[1],chapterId:c?.[1]};
  }
  return{status:'unsupported' as Status,error:`Domain ${u.hostname} belum punya adapter. Gunakan upload gambar/CBZ.`};
}
const j=async(p:string)=>{const r=await fetch(API+p,{headers:{'User-Agent':'MangaVoiceAI/1.0'}});if(!r.ok)throw new Error('MangaDex '+r.status);return r.json()};
export async function mangaInfo(id:string){
  const{data:m}=await j(`/manga/${id}?includes[]=cover_art`);
  const cover=m.relationships.find((r:any)=>r.type==='cover_art')?.attributes?.fileName;
  const a=m.attributes;
  return{id,title:a.title.en||Object.values(a.title)[0],description:a.description?.en||'',status:a.status,
    genres:a.tags.map((t:any)=>t.attributes.name.en),cover:cover&&`https://uploads.mangadex.org/covers/${id}/${cover}.256.jpg`,lastChapter:a.lastChapter};
}
export async function chapters(id:string,lang='en'){
  const out:any[]=[];
  for(let o=0;;o+=500){const r=await j(`/manga/${id}/feed?translatedLanguage[]=${lang}&order[chapter]=asc&limit=500&offset=${o}`);
    out.push(...r.data);if(o+500>=r.total)break}
  const seen=new Set<string>();
  return out.filter(c=>c.attributes.chapter&&c.attributes.pages>0).map(c=>({id:c.id,number:parseFloat(c.attributes.chapter),title:c.attributes.title}))
    .sort((a,b)=>a.number-b.number).filter(c=>!seen.has(c.number+'')&&seen.add(c.number+''));
}
export async function pageUrls(chapterId:string){
  const r=await j(`/at-home/server/${chapterId}`);
  return r.chapter.data.map((f:string)=>`${r.baseUrl}/data/${r.chapter.hash}/${f}`);
}
